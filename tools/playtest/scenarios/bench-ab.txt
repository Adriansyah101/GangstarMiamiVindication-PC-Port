# STREAM_VBO's A/B in one run (dev): bench-city's settings, 120 s standing still in the city while STREAM_VBO turns
# on and off by turns each PERF_PROFILE window (GMV_STREAM_AB=1 in the environment), so that both see the same traffic.
# The [perf] STREAM_AB lines say which state each window had.
dev
env GMV_STREAM_AB=1
ini FULLSCREEN=0
ini SCREEN_WIDTH=1600
ini SCREEN_HEIGHT=900
ini SHORT_WARNING=1
ini NO_INTRO=1
ini FRAME_MS=0
ini SWAP_INTERVAL=0
ini MSAA=4
ini ANISOTROPIC_FILTERING=16
ini AMBIENT_OCCLUSION=1
ini BLOOM=2
ini WATER_REFLECTIONS=1
ini VEHICLE_REFLECTIONS=1
ini DYNAMIC_SHADOWS=1
ini SUN_MOON=1
ini SPAWN_FADE=1
ini DRAW_DISTANCE=2
ini FOG_MODE=2
ini TRAFFIC_DENSITY=3
ini PEDESTRIAN_DENSITY=3
ini PARKED_CAR_DENSITY=3
ini KEY_MOVE=1
ini PERF_PROFILE=1
save Gangstar2.sav
wait 8
key SPACE
wait 12
click 0.13 0.39
wait 45
shot still-start
wait 120
shot still-end
expectlog ^\[perf\]   STREAM_AB: this window STREAM_VBO=0
expectlog ^\[perf\]   STREAM_AB: this window STREAM_VBO=1
